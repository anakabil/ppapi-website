    // members/{uid}

    // Catatan anggaran evaluasi (diverifikasi pada proyek nyata, 2026-10-15): Firestore
    // membatasi 1.000 ekspresi per permintaan. Karena itu, pada PEMBARUAN hanya bagian
    // dokumen yang BERUBAH yang divalidasi (affectedKeys), dan pemeriksaan teks memakai
    // get() + strMax (lebih hemat daripada rangkaian "in"/"[]"). Bagian yang tidak
    // berubah sudah tervalidasi saat ditulis sebelumnya.

    function strMax(v, max) {
      return v == null || v == '' || (v is string && v.size() <= max);
    }
    function strOrTime(v, max) {
      return strMax(v, max) || v is timestamp;
    }
    function listMax(v, max) {
      return v == null || (v is list && v.size() <= max);
    }

    // profile: hanya 12 kolom yang dipakai situs (sama dengan store.js), teks dibatasi panjangnya.
    function validProfile(p) {
      return p is map
          && p.keys().hasOnly([
               'birthPlace', 'birthDate', 'gender', 'city', 'province', 'address',
               'company', 'position', 'experienceYears', 'education', 'bio', 'linkedin'
             ])
          && strMax(p.get('birthPlace', ''), 300) && strMax(p.get('birthDate', ''), 300)
          && strMax(p.get('gender', ''), 300) && strMax(p.get('city', ''), 300)
          && strMax(p.get('province', ''), 300) && strMax(p.get('address', ''), 2000)
          && strMax(p.get('company', ''), 300) && strMax(p.get('position', ''), 300)
          && strMax(p.get('education', ''), 300) && strMax(p.get('bio', ''), 2000)
          && strMax(p.get('linkedin', ''), 300)
          && (p.get('experienceYears', null) == null
              || (p.experienceYears is int && p.experienceYears >= 0 && p.experienceYears <= 80));
    }

    function validPembayaran(p) {
      return p is map && p.size() <= 3
          && strMax(p.get('metode', ''), 60)
          && strMax(p.get('tanggal', ''), 10)
          && strMax(p.get('buktiUrl', ''), 1000);
    }
    function validPernyataan(p) {
      return p is map && p.size() <= 5
          && p.get('kodeEtik', false) is bool
          && p.get('kebenaranData', false) is bool
          && p.get('privasi', false) is bool
          && strMax(p.get('namaTerang', ''), 120)
          && strMax(p.get('tanggal', ''), 10);
    }

    // Formulir pendaftaran (application) — batas sama dengan store.js:
    // pengalaman ≤ 10, dokumen ≤ 12, catatan ≤ 1.000. Pas foto disimpan di
    // member_photos/{uid} (pasFotoAda = ada/tidak); pasFoto di sini hanya untuk data lama.
    function validApplication(a) {
      return a is map
          && a.keys().hasOnly([
               'type', 'kategori', 'submittedAt', 'updatedAt', 'pengalaman', 'dokumen',
               'pasFoto', 'pasFotoAda', 'pembayaran', 'pernyataan', 'catatan'
             ])
          && a.get('type', 'baru') in ['baru', 'perpanjangan']
          && a.get('kategori', '') in ['', 'Praktisi', 'Ahli', 'Mitra Korporat', 'Kehormatan']
          && strOrTime(a.get('submittedAt', ''), 40)
          && strOrTime(a.get('updatedAt', ''), 40)
          && listMax(a.get('pengalaman', null), 10)
          && listMax(a.get('dokumen', null), 12)
          && (a.get('pasFoto', '') == '' || a.get('pasFoto', '') == null || jpegDataUrl(a.pasFoto))
          && a.get('pasFotoAda', false) is bool
          && validPembayaran(a.get('pembayaran', {}))
          && validPernyataan(a.get('pernyataan', {}))
          && strMax(a.get('catatan', ''), 1000);
    }

    // Bagian inti dokumen anggota (murah): kolom yang diizinkan, uid, kategori, status.
    function memberCore(d, uid) {
      return d.keys().hasOnly([
               'uid', 'name', 'email', 'phone', 'photoURL', 'kta', 'memberType', 'status',
               'joinedAt', 'validUntil', 'approvedAt', 'approvedBy', 'profile', 'competencies',
               'certificates', 'application', 'review', 'contoh', 'createdAt', 'updatedAt'
             ])
          && d.uid == uid
          && d.memberType in ['Praktisi', 'Ahli', 'Mitra Korporat', 'Kehormatan']
          && d.status in ['Draf', 'Menunggu Verifikasi', 'Perlu Perbaikan', 'Aktif', 'Ditolak', 'Nonaktif'];
    }

    // Kolom teks & waktu (dipakai saat pembuatan dokumen dan penulisan oleh admin).
    function memberStrings(d) {
      return strMax(d.get('name', ''), 120)
          && strMax(d.get('email', ''), 200)
          && strMax(d.get('phone', ''), 24)
          && strMax(d.get('photoURL', ''), 1000)
          && strMax(d.get('kta', ''), 40)
          && strMax(d.get('joinedAt', ''), 32)
          && strMax(d.get('validUntil', ''), 10)
          && strMax(d.get('approvedAt', ''), 40)
          && strMax(d.get('approvedBy', ''), 128)
          && strOrTime(d.get('createdAt', ''), 40)
          && strOrTime(d.get('updatedAt', ''), 40)
          && (d.get('contoh', false) is bool);
    }

    // Validasi lengkap (pembuatan dokumen & penulisan admin): bagian yang ada saja.
    function validMember(d, uid) {
      return memberCore(d, uid)
          && memberStrings(d)
          && (!('profile' in d) || validProfile(d.profile))
          && listMax(d.get('competencies', null), 50)
          && listMax(d.get('certificates', null), 50)
          && (!('application' in d) || validApplication(d.application))
          && (!('review' in d) || d.review is map);
    }

    // Email/nomor HP pada data anggota baru harus sama dengan milik akun yang masuk
    // (atau kosong). Email hanya diterima bila sudah diverifikasi pemiliknya.
    function contactMatchesAccount(d) {
      return (d.get('email', '') == '' || d.get('email', '') == null || verifiedEmail(d.email))
          && (d.get('phone', '') == '' || d.get('phone', '') == null
                || ('phone_number' in request.auth.token && d.phone == request.auth.token.phone_number));
    }

    // Formulir sudah diajukan (application.submittedAt terisi: waktu server, atau teks data lama).
    function submitted(d) {
      return d.get('application', {}).get('submittedAt', '') is timestamp
          || d.get('application', {}).get('submittedAt', '') != '';
    }

    // Boleh mengajukan perpanjangan: Nonaktif, atau Aktif dengan masa berlaku yang
    // telah habis / berakhir dalam ±60 hari.
    function renewable(m) {
      return m.status == 'Nonaktif'
          || (m.status == 'Aktif' && isoDate(m.get('validUntil', ''))
              && dateTs(m.validUntil) <= request.time + duration.value(61, 'd'));
    }

    // Pengajuan lengkap: kategori yang boleh diajukan (Kehormatan hanya untuk perpanjangan
    // oleh pemegangnya), nama, ketiga pernyataan, dan nama terang.
    function completeSubmission(before, after) {
      let a = after.get('application', {});
      let p = a.get('pernyataan', {});
      let kategori = a.get('kategori', '');
      return (kategori in ['Praktisi', 'Ahli', 'Mitra Korporat']
                || (kategori != '' && kategori == before.get('memberType', '')
                    && before.get('approvedAt', '') != ''))
          && after.get('name', '') is string && after.get('name', '').size() >= 3
          && p.get('kodeEtik', false) == true
          && p.get('kebenaranData', false) == true
          && p.get('privasi', false) == true
          && p.get('namaTerang', '') is string && p.get('namaTerang', '').size() >= 3;
    }

    // Perubahan status yang boleh dilakukan ANGGOTA sendiri:
    //   Draf | Perlu Perbaikan | Ditolak  → Menunggu Verifikasi   (mengajukan formulir lengkap)
    //   Aktif (habis/≤60 hari) | Nonaktif → Menunggu Verifikasi   (perpanjangan, lengkap)
    //   Menunggu Verifikasi              → Draf                   (menarik pengajuan)
    function ownerTransition(before, after) {
      return (before.status in ['Draf', 'Perlu Perbaikan', 'Ditolak']
                && after.status == 'Menunggu Verifikasi' && submitted(after)
                && completeSubmission(before, after))
          || (renewable(before) && after.status == 'Menunggu Verifikasi' && submitted(after)
                && after.get('application', {}).get('type', '') == 'perpanjangan'
                && completeSubmission(before, after))
          || (before.status == 'Menunggu Verifikasi' && after.status == 'Draf' && !submitted(after));
    }

    // Formulir boleh diubah anggota hanya saat belum/tidak sedang diverifikasi.
    function applicationEditable(before, after) {
      return before.status in ['Draf', 'Perlu Perbaikan', 'Ditolak']
          || renewable(before)
          || (before.status == 'Menunggu Verifikasi' && (after.status == 'Draf' || !submitted(before)));
    }

    // Kategori yang diajukan anggota: Praktisi, Ahli, atau Mitra Korporat. Kategori lain
    // (Kehormatan) hanya bila sama dengan kategori resmi anggota atau tidak berubah.
    function kategoriAllowed(after, before) {
      let k = after.get('application', {}).get('kategori', '');
      return k in ['', 'Praktisi', 'Ahli', 'Mitra Korporat']
          || k == before.get('memberType', '')
          || k == before.get('application', {}).get('kategori', '');
    }

    // Waktu pada formulir yang ditulis anggota = waktu server (atau tidak berubah /
    // dikosongkan saat menarik pengajuan). Antrean verifikasi diurutkan menurut submittedAt.
    function appTimesOk(after, before) {
      let a = after.get('application', {});
      let b = before.get('application', {});
      return (a.get('updatedAt', '') == b.get('updatedAt', '') || a.get('updatedAt', '') == request.time)
          && (a.get('submittedAt', '') == b.get('submittedAt', '') || a.get('submittedAt', '') == ''
              || a.get('submittedAt', '') == request.time);
    }

    // Selama "Menunggu Verifikasi" (formulir sudah diajukan), data yang sedang diperiksa
    // sekretariat tidak berubah kecuali pengajuan ditarik (→ Draf).
    function reviewedDataKept(changed, before, after) {
      return before.status != 'Menunggu Verifikasi' || !submitted(before) || after.status == 'Draf'
          || !changed.hasAny(['name', 'phone', 'photoURL', 'profile', 'competencies', 'certificates']);
    }

    // Setelah pernah disetujui (approvedAt terisi), nama yang tercantum pada KTA tidak
    // diubah anggota sendiri (koreksi nama oleh sekretariat). Nama kosong boleh dilengkapi.
    function identityKept(after, before) {
      return before.get('approvedAt', '') == '' || before.get('approvedAt', '') == null
          || !(before.get('name', '') is string) || before.get('name', '').size() < 3
          || after.get('name', '') == before.get('name', '');
    }

    // Pembaruan oleh anggota: hanya kolom yang diizinkan; kolom sekretariat tetap; hanya
    // bagian yang berubah yang divalidasi (hemat anggaran evaluasi).
    function ownerUpdate(after, before, uid) {
      let changed = after.diff(before).affectedKeys();
      return changed.hasOnly([
               'name', 'email', 'phone', 'photoURL', 'profile', 'competencies',
               'certificates', 'application', 'status', 'updatedAt'
             ])
          && memberCore(after, uid)
          && (!changed.hasAny(['name']) || (strMax(after.get('name', ''), 120) && identityKept(after, before)))
          && (!changed.hasAny(['email']) || after.get('email', '') == '' || after.get('email', '') == null
                || verifiedEmail(after.email))
          && (!changed.hasAny(['phone']) || strMax(after.get('phone', ''), 24))
          && (!changed.hasAny(['photoURL']) || strMax(after.get('photoURL', ''), 1000))
          && (!changed.hasAny(['profile']) || validProfile(after.profile))
          && (!changed.hasAny(['competencies']) || listMax(after.get('competencies', null), 50))
          && (!changed.hasAny(['certificates']) || listMax(after.get('certificates', null), 50))
          && (!changed.hasAny(['application'])
                || (validApplication(after.application) && applicationEditable(before, after)
                    && kategoriAllowed(after, before) && appTimesOk(after, before)))
          && (!changed.hasAny(['updatedAt']) || after.updatedAt == request.time)
          && (!changed.hasAny(['status']) || ownerTransition(before, after))
          && reviewedDataKept(changed, before, after);
    }

    // Pembaruan oleh admin: bagian yang berubah divalidasi; kolom inti selalu diperiksa.
    function adminUpdate(after, before, uid) {
      let changed = after.diff(before).affectedKeys();
      return memberCore(after, uid)
          && memberStrings(after)
          && (!changed.hasAny(['profile']) || !('profile' in after) || validProfile(after.profile))
          && (!changed.hasAny(['competencies']) || listMax(after.get('competencies', null), 50))
          && (!changed.hasAny(['certificates']) || listMax(after.get('certificates', null), 50))
          && (!changed.hasAny(['application']) || !('application' in after) || validApplication(after.application))
          && (!changed.hasAny(['review']) || !('review' in after) || after.review is map);
    }

    match /members/{uid} {
      allow get: if isOwner(uid) || isAdmin();
      allow list: if isAdmin();

      // Akun baru: status "Draf", kategori awal "Praktisi", tanpa nomor KTA, tanpa
      // masa berlaku/persetujuan/catatan sekretariat, formulir belum diajukan.
      allow create: if (isOwner(uid)
                          && validMember(request.resource.data, uid)
                          && contactMatchesAccount(request.resource.data)
                          && request.resource.data.status == 'Draf'
                          && request.resource.data.memberType == 'Praktisi'
                          && request.resource.data.get('kta', '') == ''
                          && request.resource.data.get('validUntil', '') == ''
                          && request.resource.data.get('approvedAt', '') == ''
                          && request.resource.data.get('approvedBy', '') == ''
                          && !('contoh' in request.resource.data)
                          && request.resource.data.get('review', {}).get('note', '') == ''
                          && request.resource.data.get('review', {}).get('history', []).size() == 0
                          && !submitted(request.resource.data)
                          && request.resource.data.get('application', {}).get('kategori', '')
                               in ['', 'Praktisi', 'Ahli', 'Mitra Korporat']
                          && isoDate(request.resource.data.joinedAt))
                    || (isAdmin() && validMember(request.resource.data, uid));

      allow update: if (isOwner(uid) && ownerUpdate(request.resource.data, resource.data, uid))
                    || (isAdmin() && adminUpdate(request.resource.data, resource.data, uid));

      // Data anggota tidak dihapus dari situs (penghapusan atas permintaan pemilik data
      // dilakukan sekretariat melalui Firebase Console).
      allow delete: if false;
    }
